# hive Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `hive`, a Rust ratatui TUI that SSHes into multiple machines (password or key), shows a live CPU/RAM/load monitor per node, runs ad-hoc commands, and launches MPI jobs across the cluster — all driven from a YAML config.

**Architecture:** A Tokio runtime hosts the app: background tasks keep one `russh` SSH session per machine, poll metrics on a timer, and execute commands; a separate UI task runs the ratatui event loop and communicates with the rest via `tokio::mpsc` channels. Config and metrics parsing are pure, unit-tested modules; the TUI is message-driven (`Action` in, `UiEvent` out).

**Tech Stack:** Rust 2021, Tokio, russh 0.4, ratatui 0.29, crossterm 0.28, serde / serde_yaml 0.9, thiserror, anyhow, clap 4, chrono.

```rust
use crate::error::HiveError;
use serde::{Deserialize, Serialize};
use std::path::Path;

#[derive(Debug, Clone, Serialize, Deserialize)]
#[serde(tag = "method", rename_all = "lowercase")]
pub enum Auth {
    Password { password: String },
    Key { key_path: String },
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct MachineConfig {
    pub name: String,
    pub host: String,
    #[serde(default = "default_port")]
    pub port: u16,
    pub user: String,
    pub auth: Auth,
    #[serde(default)]
    pub tags: Vec<String>,
}

fn default_port() -> u16 { 22 }

#[derive(Debug, Clone, Serialize, Deserialize, Default)]
pub struct MpiConfig {
    #[serde(default = "default_launcher")]
    pub launcher: String,
    #[serde(default)]
    pub default_args: String,
}
fn default_launcher() -> String { "mpirun".into() }

#[derive(Debug, Clone, Serialize, Deserialize, Default)]
pub struct ClusterConfig {
    #[serde(default = "default_poll")]
    pub poll_interval_ms: u64,
    #[serde(default)]
    pub mpi: MpiConfig,
    #[serde(default)]
    pub machines: Vec<MachineConfig>,
}

fn default_poll() -> u64 { 2000 }

impl ClusterConfig {
    pub fn load(path: &Path) -> Result<Self, HiveError> {
        if !path.exists() {
            return Ok(ClusterConfig::default());
        }
        let text = std::fs::read_to_string(path)
            .map_err(|e| HiveError::Config(format!("read {}: {e}", path.display())))?;
        let cfg: ClusterConfig = serde_yaml::from_str(&text)
            .map_err(|e| HiveError::Config(format!("parse {}: {e}", path.display())))?;
        cfg.validate()?;
        Ok(cfg)
    }

    pub fn save(&self, path: &Path) -> Result<(), HiveError> {
        let text = serde_yaml::to_string(self)
            .map_err(|e| HiveError::Config(format!("serialize: {e}")))?;
        if let Some(parent) = path.parent() {
            std::fs::create_dir_all(parent).ok();
        }
        std::fs::write(path, text)
            .map_err(|e| HiveError::Config(format!("write {}: {e}", path.display())))?;
        // restrict permissions when we created it
        #[cfg(unix)]
        {
            use std::os::unix::fs::PermissionsExt;
            let _ = std::fs::set_permissions(path, std::fs::Permissions::from_mode(0o600));
        }
        Ok(())
    }

    fn validate(&self) -> Result<(), HiveError> {
        for m in &self.machines {
            match &m.auth {
                Auth::Password { password } if password.is_empty() =>
                    return Err(HiveError::Config(format!("{}: empty password", m.name))),
                Auth::Key { key_path } if key_path.is_empty() =>
                    return Err(HiveError::Config(format!("{}: empty key_path", m.name))),
                _ => {}
            }
        }
        Ok(())
    }
}
```

```rust
#[derive(Debug, Clone, Default, PartialEq)]
pub struct MachineStats {
    pub cores: u32,
    pub cpu_percent: f32,   // 0..100, busy
    pub mem_used_mib: u64,
    pub mem_total_mib: u64,
    pub load1: f32,
    pub load5: f32,
    pub load15: f32,
    pub uptime_secs: f64,
}

pub fn parse_nproc(s: &str) -> anyhow::Result<u32> {
    Ok(s.trim().parse()?)
}

pub fn parse_free_m(s: &str) -> anyhow::Result<(u64, u64)> {
    // find the Mem: line; use "available" to compute used = total - available
    for line in s.lines() {
        if line.starts_with("Mem:") {
            let cols: Vec<&str> = line.split_whitespace().collect();
            // cols: ["Mem:", total, used, free, shared, buff/cache, available]
            let total: u64 = cols[1].parse()?;
            if cols.len() >= 7 {
                let avail: u64 = cols[6].parse()?;
                return Ok((total - avail, total));
            }
            let used: u64 = cols[2].parse()?;
            return Ok((used, total));
        }
    }
    anyhow::bail!("no Mem: line in free output")
}

pub fn parse_loadavg(s: &str) -> anyhow::Result<(f32, f32, f32)> {
    let cols: Vec<&str> = s.split_whitespace().collect();
    Ok((cols[0].parse()?, cols[1].parse()?, cols[2].parse()?))
}

pub fn parse_mpstat(s: &str) -> anyhow::Result<f32> {
    // last numeric line carries %idle in final column
    let line = s.lines().filter(|l| l.contains('%')).last()
        .ok_or_else(|| anyhow::anyhow!("no mpstat data"))?;
    let cols: Vec<&str> = line.split_whitespace().collect();
    let idle: f32 = cols.last().ok_or_else(|| anyhow::anyhow!("empty"))?.parse()?;
    Ok(100.0 - idle)
}

pub fn parse_uptime(s: &str) -> anyhow::Result<f64> {
    // /proc/uptime first field
    let first = s.split_whitespace().next().ok_or_else(|| anyhow::anyhow!("empty"))?;
    Ok(first.parse()?)
}
```

```rust
use crate::config::{Auth, MachineConfig};
use crate::error::{HiveError, Result};
use russh::client::{self, Handle};
use russh::keys::load_secret_key;
use std::sync::Arc;

pub struct Client;

impl client::Handler for Client {
    type Error = russh::Error;
    async fn check_host_key(
        &mut self,
        _host: &str,
        _port: &mut u16,
        _key: &russh::keys::PublicKey,
    ) -> Result<bool, Self::Error> {
        Ok(true) // lab environment: accept unknown hosts
    }
}

#[derive(Debug, Clone)]
pub struct CmdOutput {
    pub stdout: String,
    pub stderr: String,
    pub exit: u32,
}

pub struct Session {
    handle: Handle<Client>,
    pub name: String,
}

impl Session {
    pub async fn connect(cfg: &MachineConfig) -> Result<Self> {
        let config = Arc::new(russh::client::Config::default());
        let mut handle = client::connect(config, (cfg.host.as_str(), cfg.port), Client)
            .await
            .map_err(HiveError::Ssh)?;
        let ok = match &cfg.auth {
            Auth::Password { password } => {
                handle.auth_password(&cfg.user, password).await.map_err(HiveError::Ssh)?
            }
            Auth::Key { key_path } => {
                let expanded = shellexpand::full(key_path)
                    .map(|s| s.into_owned())
                    .unwrap_or_else(|_| key_path.clone());
                let key = load_secret_key(expanded, None).await.map_err(HiveError::Ssh)?;
                handle.auth_publickey(&cfg.user, Arc::new(key)).await.map_err(HiveError::Ssh)?
            }
        };
        if !ok.success() {
            return Err(HiveError::Auth(cfg.name.clone()));
        }
        Ok(Session { handle, name: cfg.name.clone() })
    }

    pub async fn exec(&mut self, cmd: &str) -> Result<CmdOutput> {
        let mut chan = self.handle.channel_open_session().await.map_err(HiveError::Ssh)?;
        chan.exec(true, cmd).await.map_err(HiveError::Ssh)?;
        let mut out = CmdOutput { stdout: String::new(), stderr: String::new(), exit: 0 };
        while let Some(msg) = chan.wait().await {
            use russh::ChannelMsg::*;
            match msg {
                Data { data } => out.stdout.push_str(&String::from_utf8_lossy(&data)),
                ExtendedData { data, ext } if ext == 1 =>
                    out.stderr.push_str(&String::from_utf8_lossy(&data)),
                ExitStatus { exit_status } => out.exit = exit_status,
                _ => {}
            }
        }
        Ok(out)
    }
}
```

```rust
use crate::config::ClusterConfig;
use crate::error::{HiveError, Result};
use crate::ssh::Session;

#[derive(Debug, Clone)]
pub struct Worker {
    pub host: String,
    pub slots: u32,
}

/// One `host:slots` line per worker, as used by mpirun -hostfile.
pub fn build_hostfile(workers: &[Worker]) -> String {
    let mut s = String::new();
    for w in workers {
        s.push_str(&format!("{}:{}\n", w.host, w.slots));
    }
    s
}

/// Run a command on an already-open session, return full output.
pub async fn run_on(session: &mut Session, cmd: &str) -> Result<String> {
    let o = session.exec(cmd).await?;
    if o.exit != 0 {
        return Err(HiveError::Command {
            host: session.name.clone(),
            code: o.exit as i32,
            stderr: o.stderr,
        });
    }
    Ok(o.stdout)
}

/// Launch an MPI job from `head` across `workers`.
/// Assumes mpirun/mpiexec + the target binary already exist on the nodes.
pub async fn dispatch_mpi(
    head: &mut Session,
    cfg: &ClusterConfig,
    workers: &[Worker],
    binary: &str,
    args: &str,
) -> Result<String> {
    let hf = build_hostfile(workers);
    // write hostfile to head node
    let remote = "/tmp/hive_hostfile";
    let put = format!("cat > {remote} <<'EOF'\n{hf}EOF");
    head.exec(&put).await?;
    let launcher = &cfg.mpi.launcher;
    let defaults = &cfg.mpi.default_args;
    let cmd = format!(
        "{launcher} -hostfile {remote} {defaults} {args} {binary}"
    );
    let o = head.exec(&cmd).await?;
    Ok(format!("exit={}\n{}\n{}", o.exit, o.stdout, o.stderr))
}
```

```rust
use crate::app::{Action, ConnStatus, UiEvent};
use crate::config::{ClusterConfig, MachineConfig};
use crate::error::Result;
use crate::jobs::{dispatch_mpi, run_on, Worker};
use crate::metrics::{MachineStats, parse_free_m, parse_loadavg, parse_mpstat, parse_nproc, parse_uptime};
use crate::ssh::Session;
use std::collections::HashMap;
use std::sync::Arc;
use tokio::sync::mpsc::UnboundedSender;

type Ui = UnboundedSender<UiEvent>;

async fn collect_stats(session: &mut Session) -> Result<MachineStats> {
    let cores = parse_nproc(&session.exec("nproc").await?.stdout)?;
    let (mem_used, mem_total) = parse_free_m(&session.exec("free -m").await?.stdout)?;
    let (load1, load5, load15) = parse_loadavg(&session.exec("cat /proc/loadavg").await?.stdout)?;
    let cpu_idle = match session.exec("mpstat 1 1 2>/dev/null || top -bn1").await {
        Ok(o) => parse_mpstat(&o.stdout).unwrap_or(100.0),
        Err(_) => 100.0,
    };
    let uptime = parse_uptime(&session.exec("cat /proc/uptime").await?.stdout)?;
    Ok(MachineStats {
        cores,
        cpu_percent: 100.0 - cpu_idle,
        mem_used_mib: mem_used,
        mem_total_mib: mem_total,
        load1, load5, load15,
        uptime_secs: uptime,
    })
}

/// Spawns a per-machine session, then polls stats on an interval.
async fn spawn_machine(mc: MachineConfig, ui: Ui, poll_ms: u64) {
    ui.send(UiEvent::Conn { name: mc.name.clone(), status: ConnStatus::Connecting }).ok();
    let mut session = match Session::connect(&mc).await {
        Ok(s) => s,
        Err(e) => {
            ui.send(UiEvent::Conn { name: mc.name.clone(), status: ConnStatus::Error(e.to_string()) }).ok();
            return;
        }
    };
    ui.send(UiEvent::Conn { name: mc.name.clone(), status: ConnStatus::Connected }).ok();
    loop {
        match collect_stats(&mut session).await {
            Ok(stats) => ui.send(UiEvent::Stats { name: mc.name.clone(), stats }).ok(),
            Err(e) => ui.send(UiEvent::Log(format!("{} stats err: {e}", mc.name))).ok(),
        }
        tokio::time::sleep(std::time::Duration::from_millis(poll_ms)).await;
    }
}

/// Handles user actions (connect / run / mpi).
pub async fn run_dispatcher(
    mut rx: tokio::sync::mpsc::UnboundedReceiver<Action>,
    ui: Ui,
    cfg: Arc<ClusterConfig>,
) {
    // open map of live sessions for run/mpi (name -> Session)
    let mut sessions: HashMap<String, Session> = HashMap::new();
    while let Some(action) = rx.recv().await {
        match action {
            Action::Quit => break,
            Action::ConnectAll | Action::Connect(_) => {
                let poll = cfg.poll_interval_ms;
                for mc in &cfg.machines {
                    let need = match &action {
                        Action::Connect(n) => &mc.name == n,
                        _ => true,
                    };
                    if need {
                        let ui2 = ui.clone();
                        let mc2 = mc.clone();
                        tokio::spawn(async move { spawn_machine(mc2, ui2, poll).await });
                    }
                }
            }
            Action::Disconnect(name) => { sessions.remove(&name); }
            Action::SaveConfig(c) => {
                if let Err(e) = c.save(std::path::Path::new(&std::env::var("HIVE_CONFIG").unwrap_or_else(|_| "cluster.yaml".into()))) {
                    ui.send(UiEvent::Log(format!("save failed: {e}"))).ok();
                } else {
                    ui.send(UiEvent::Log("config saved".into())).ok();
                }
            }
            Action::Run { targets, cmd } => {
                for name in targets {
                    if let Some(mc) = cfg.machines.iter().find(|m| m.name == name) {
                        match Session::connect(mc).await {
                            Ok(mut s) => match run_on(&mut s, &cmd).await {
                                Ok(out) => ui.send(UiEvent::RunOutput { name, out }).ok(),
                                Err(e) => ui.send(UiEvent::Log(format!("{name}: {e}"))).ok(),
                            },
                            Err(e) => ui.send(UiEvent::Log(format!("{name} connect: {e}"))).ok(),
                        }
                    }
                }
            }
            Action::LaunchMpi { head, workers, binary, args } => {
                let head_cfg = match cfg.machines.iter().find(|m| m.name == head) {
                    Some(c) => c.clone(),
                    None => { ui.send(UiEvent::Log(format!("head {head} not found"))).ok(); continue; }
                };
                match Session::connect(&head_cfg).await {
                    Ok(mut s) => {
                        let ws: Vec<Worker> = workers.iter().filter_map(|n| {
                            cfg.machines.iter().find(|m| &m.name == n).map(|m| Worker {
                                host: m.host.clone(),
                                slots: 4, // placeholder; refine via metrics later
                            })
                        }).collect();
                        match dispatch_mpi(&mut s, &cfg, &ws, &binary, &args).await {
                            Ok(out) => ui.send(UiEvent::MpiOutput(out)).ok(),
                            Err(e) => ui.send(UiEvent::MpiOutput(format!("ERR: {e}"))).ok(),
                        }
                    }
                    Err(e) => ui.send(UiEvent::MpiOutput(format!("head connect: {e}"))).ok(),
                }
            }
        }
    }
}
```

```rust
pub mod machines;
pub mod monitor;
pub mod run;
pub mod mpi;
pub mod logs;

use crate::app::{Action, AppState, Tab, UiEvent};
use crate::runner;
use crossterm::event::{self, Event, KeyCode};
use ratatui::backend::CrosstermBackend;
use ratatui::Terminal;
use std::io;
use std::time::Duration;
use tokio::sync::mpsc::{unbounded_channel, UnboundedSender};

pub fn run(app_state: &mut AppState, cfg: std::sync::Arc<crate::config::ClusterConfig>) -> io::Result<()> {
    // build initial machine views
    for m in &cfg.machines {
        app_state.machines.push(crate::app::MachineView {
            name: m.name.clone(), host: m.host.clone(),
            status: crate::app::ConnStatus::Disconnected, tags: m.tags.clone(),
        });
    }

    let (ui_tx, mut ui_rx) = unbounded_channel::<UiEvent>();
    let (action_tx, action_rx) = unbounded_channel::<Action>();
    let rt = tokio::runtime::Runtime::new()?;
    rt.block_on(async {
        tokio::spawn(runner::run_dispatcher(action_rx, ui_tx.clone(), cfg.clone()));
        // input reader thread
        let input_tx = action_tx.clone();
        std::thread::spawn(move || loop {
            if event::poll(Duration::from_millis(100)).unwrap_or(false) {
                if let Ok(Event::Key(k)) = event::read() {
                    if k.code == KeyCode::Char('q') {
                        input_tx.send(Action::Quit).ok();
                        break;
                    }
                }
            }
        });

        let mut terminal = Terminal::new(CrosstermBackend::new(io::stdout()))?;
        loop {
            // drain ui events
            while let Ok(ev) = ui_rx.try_recv() {
                apply_event(app_state, ev);
            }
            terminal.draw(|f| ui(app_state, f))?;
            // also check for quit
            if app_state.quit { break; }
            tokio::time::sleep(Duration::from_millis(50)).await;
        }
        Ok::<(), io::Error>(())
    })?;

    // NOTE: action_tx is unused here beyond input thread; keep reference to avoid drop
    let _ = &action_tx;
    disable_raw_mode()?;
    Ok(())
}

use crossterm::terminal::{disable_raw_mode, enable_raw_mode};
use crossterm::execute;
use crossterm::terminal::{EnterAlternateScreen, LeaveAlternateScreen};

fn apply_event(state: &mut AppState, ev: UiEvent) {
    use crate::app::ConnStatus;
    match ev {
        UiEvent::Conn { name, status } => {
            if let Some(m) = state.machines.iter_mut().find(|x| x.name == name) { m.status = status; }
        }
        UiEvent::Stats { name, stats } => { state.stats.insert(name, stats); }
        UiEvent::Log(msg) => state.logs.push(crate::app::LogEntry {
            ts: chrono::Local::now().format("%H:%M:%S").to_string(), msg,
        }),
        UiEvent::RunOutput { name, out } => state.run_output.push_str(&format!("[{name}]\n{out}\n")),
        UiEvent::MpiOutput(out) => state.mpi_output.push_str(&out),
    }
}

fn ui(state: &AppState, f: &mut ratatui::Frame) {
    use ratatui::layout::{Constraint, Direction, Layout};
    use ratatui::widgets::{Block, Borders, Tabs};
    let tabs = ["Machines", "Monitor", "Run", "MPI", "Logs"];
    let idx = match state.tab { Tab::Machines=>0, Tab::Monitor=>1, Tab::Run=>2, Tab::Mpi=>3, Tab::Logs=>4 };
    let chunks = Layout::default().direction(Direction::Vertical)
        .constraints([Constraint::Length(1), Constraint::Min(0)]).split(f.area());
    let t = Tabs::new(tabs.iter().map(|s| ratatui::text::Line::from(*s)).collect::<Vec<_>>())
        .select(idx).block(Block::default().borders(Borders::ALL));
    f.render_widget(t, chunks[0]);
    match state.tab {
        Tab::Machines => machines::render(state, f, chunks[1]),
        Tab::Monitor => monitor::render(state, f, chunks[1]),
        Tab::Run => run::render(state, f, chunks[1]),
        Tab::Mpi => mpi::render(state, f, chunks[1]),
        Tab::Logs => logs::render(state, f, chunks[1]),
    }
}
```
